SQL QUERY RUNNER // PRODUCTION NODE 01 READ COMMITTED 
-- Semantic Alumni Discovery via pgvector HNSW
SELECT 
    a.alumni_id,
    a.full_name,
    a.current_company,
    a.verified_role,
    1 - (a.profile_embedding <=> q.query_vector) AS cosine_similarity
FROM 
    institutional_alumni a,
    (
        SELECT gemini_embed_text(
            'Distributed systems, SRE, fault tolerance'
        )::vector(384) AS query_vector
    ) q
WHERE 
    a.referral_quota_active = TRUE
    AND a.verified_status = 'ACCREDITED'
ORDER BY 
    a.profile_embedding <=> q.query_vector ASC
LIMIT 5;
INDEX: idx_alumni_hnsw TOTAL EXECUTION: 6.84ms
